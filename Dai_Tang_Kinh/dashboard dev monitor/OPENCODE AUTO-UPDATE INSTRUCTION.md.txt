def get_gdrive_service():

    """Get authenticated Google Drive service"""

    # Use Application Default Credentials

    credentials = service_account.Credentials.from_service_account_file(

        'config/service-account.json'  # Your service account key file

    )

    return build('drive', 'v3', credentials=credentials)

def download_instruction():

    """Download latest instruction from Google Drive"""

    service = get_gdrive_service()

    

    request = service.files().get_media(fileId=FILE_ID)

    fh = BytesIO()

    downloader = MediaIoBaseDownload(fh, request)

    

    done = False

    while not done:

        status, done = downloader.next_chunk()

    

    content = fh.getvalue().decode('utf-8')

    return content

def compute_hash(content):

    """Compute SHA256 hash of content"""

    return hashlib.sha256(content.encode()).hexdigest()

def has_instruction_changed():

    """Check if instruction file changed on GDrive"""

    try:

        # Download latest

        latest_content = download_instruction()

        latest_hash = compute_hash(latest_content)

        

        # Read previous hash

        if os.path.exists(LOCAL_HASH):

            with open(LOCAL_HASH, 'r') as f:

                previous_hash = f.read().strip()

        else:

            previous_hash = None

        

        # Compare

        if latest_hash != previous_hash:

            print(f"✅ Instruction updated! (hash: {previous_hash[:8]}→{latest_hash[:8]})")

            

            # Save new content

            with open(LOCAL_INSTRUCTION, 'w', encoding='utf-8') as f:

                f.write(latest_content)

            

            # Save new hash

            with open(LOCAL_HASH, 'w') as f:

                f.write(latest_hash)

            

            return True, latest_content

        else:

            print("⏭️  Instruction unchanged")

            return False, latest_content

    

    except Exception as e:

        print(f"❌ Error syncing instruction: {e}")

        return False, None
